# Calcular el primer cuartil (25 % de los datos)
Q1 = np.percentile(X, 25)

# Calcular el tercer cuartil (75 % de los datos)
Q3 = np.percentile(X, 75)

# Calcular el Rango Intercuartílico (RIC)
# El RIC mide la dispersión de la mitad central de los datos
RIC = Q3 - Q1

# Calcular el límite inferior para detectar posibles valores atípicos
lim_inf = Q1 - 1.5 * RIC

# Calcular el límite superior para detectar posibles valores atípicos
lim_sup = Q3 + 1.5 * RIC

# Mostrar el valor del Rango Intercuartílico
print('RIC:', RIC)

# Mostrar el límite inferior
print('Límite inferior:', lim_inf)

# Mostrar el límite superior
print('Límite superior:', lim_sup)

# Medidas de distribución

# Calcular el coeficiente de asimetría de los datos
# Indica si la distribución está inclinada hacia la izquierda o la derecha
asimetria = stats.skew(X)

# Calcular la curtosis de los datos
# fisher=True hace que una distribución normal tenga curtosis igual a 0
curtosis = stats.kurtosis(X, fisher=True)

# Mostrar la asimetría redondeada a 4 decimales
print('Asimetría:', np.round(asimetria, 4))

# Mostrar la curtosis redondeada a 4 decimales
print('Curtosis:', np.round(curtosis, 4))

# Determinar el tipo de distribución según la curtosis
if curtosis > 0:
    print('Distribución Leptocúrtica')
elif curtosis < 0:
    print('Distribución Platicúrtica')
else:
    print('Distribución Mesocúrtica')

import matplotlib.pyplot as plt
import pandas as pd
import seaborn as sns

# 1. Cargamos el dataset Iris
iris = sns.load_dataset("iris")

# 2. Agrupamos y calculamos la media y la desviación típica
grouped_mean = iris.groupby("species").mean()
grouped_std = iris.groupby("species").std()

especies = grouped_mean.index
variables = grouped_mean.columns

# 3. Configuramos el gráfico
plt.figure(figsize=(10, 6))

offset = -0.15
for especie in especies:
    # Medias y desviaciones de la especie actual
    medias = grouped_mean.loc[especie]
    desviaciones = grouped_std.loc[especie]

    # Posiciones con desplazamiento para que las barras no se encabalguen
    x_positions = [i + offset for i in range(len(variables))]

    # GRAFICAMOS CON LÍNEAS DE CONEXIÓN
    plt.errorbar(
        x=x_positions,
        y=medias,
        yerr=desviaciones,
        fmt="o-",  # <--- 'o-' conecta los puntos con una línea continua
        capsize=6,  # Ancho de los topes de las barras de error
        elinewidth=2,  # Grosor de la línea de error
        markeredgewidth=2,
        label=especie.capitalize(),
    )

    offset += 0.15  # Desplazamiento para la siguiente especie

# 4. Ajustes estéticos y personalización
plt.title(
    "Media y Desviación Típica de las Variables de Iris por Especie",
    fontsize=14,
    pad=15,
)
plt.xticks(
    range(len(variables)),
    [v.replace("_", " ").title() for v in variables],
    fontsize=11,
)
plt.ylabel("Medida (cm)", fontsize=12)
plt.xlabel("Variables", fontsize=12)
plt.legend(title="Especies", fontsize=10)
plt.grid(True, axis="y", linestyle="--", alpha=0.5)

plt.tight_layout()
plt.show()
